  % ================= 左: アセンブリとリーダ =================

  \node[code] (i4) at (4mm, -45mm) {ret};

  % リーダの印
  \foreach \n in {i0, i1, i2, i4} {
    \draw[ptr] ($(\n.west)+(-5mm,0)$) -- ($(\n.west)+(-1mm,0)$);
  }

  \node[rule] at (36mm,  -6mm) {(a) 最初の命令};
  \node[rule] at (36mm, -19mm) {(b) ラベルが指す命令};

  \node[blk] (b2) at (122mm, -40mm) {addi a0, a0, -1\\j .L1};
  \node[blk] (b3) at (168mm, -40mm) {.L2:\\ret};

  \draw[flow] (b1.south west) -- node[gannot, left=1mm, pos=0.6] {真} (b2.north);
