% コマ9: ポインタと lvalue / rvalue のメモリ図
% int a; int *p; a = 10; p = &a; を実行した直後の状態
\documentclass[border=8pt]{standalone}
\input{figure-preamble}
\begin{document}
\begin{tikzpicture}[node distance=0mm]
  % ---- スタック上の2変数 ----
  \node[gannot, anchor=south] at ($(0,0)+(0,10.5mm)$)
      {a = 10; p = \&a; を実行した直後のスタック};
  \node[memcell hi, minimum width=44mm] (a) at (0,0) {a：10};
  \node[memcell, minimum width=44mm, below=0mm of a] (p) {p：a のアドレス};

  \node[addr, left=2.5mm of a] {s0 - 24};
  \node[addr, left=2.5mm of p] {s0 - 32};

  % ---- p が a を指す矢印 ----
  \draw[ptr] (p.east) .. controls +(16mm,0) and +(16mm,0) .. (a.east);
  \node[annot, anchor=west] at ($(a.east)+(15mm,-4.25mm)$)
      {p の値は a のアドレス\\（p = \&a の結果）};

  % ---- *p = 20 の書き込み ----
  \node[annot, text=TcEmph, anchor=north west] at ($(p.south west)+(-14mm,-6mm)$)
      {*p = 20; は矢印の先（= a のスロット）に 20 を書き込む};

  % ---- rvalue / lvalue の対応 ----
  \node[gannot, anchor=north west, align=left] at ($(p.south west)+(-14mm,-14mm)$)
      {値がほしいとき（rvalue）\hspace{2.3em}→ codegen()：\hspace{1.4em}a は 10、*p も 10\\
       アドレスがほしいとき（lvalue）→ codegen\_lval()：a は s0 - 24、*p は p の値（矢印の先）};
\end{tikzpicture}
\end{document}
